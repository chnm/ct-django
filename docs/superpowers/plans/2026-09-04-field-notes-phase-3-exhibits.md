# Field Notes Phase 3 (Exhibit Chapter Reader) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the exhibit reader's chrome and page templates with the Field Notes reader while rendering every piece of content the live templates render today.

**Architecture:** A pure-Python `exhibits/reader.py` derives numbering, navigation, and progress from the Wagtail page tree. `base_exhibit.html` becomes the reader chrome (top bar, control bar, Alpine contents sheet and lightbox). One section template maps the nine existing layouts onto grids; one `_plate.html` partial renders every image with its Caption / Full image controls. Two optional model fields (phase, related records) are added by migration.

**Tech Stack:** Django 6, Wagtail 7 (`Page.get_context`, `{% image %}`, `ParentalManyToManyField`), Alpine 3 (already loaded), Tailwind 3.4 via django-tailwind, pytest-django.

**Spec:** `docs/superpowers/specs/2026-09-04-field-notes-phase-3-exhibits-design.md`

## Global Constraints

- **Content parity.** Every row of the spec's parity table must hold: title, body, single image + caption, first/second column images + captions, comparison sliders (+ both captions), `display_title`, `link_to_subsection` ("Skip to … →"), `hero_text`, chapter `body`, `citation_text`, chapter `image` + `image_caption`. Nothing is removed from the models.
- **Tree order is the truth.** Sections = `home.get_children().live().specific()` that are `ExhibitPage`, in tree order. `link_to_next_page` / `link_to_previous_page` are never read.
- Images `object-contain` on `bg-[#2b2620]`, except the cover (`object-cover`). `*_zoomable` ignored.
- Exit and "Take me to the beginning" go to `/#exhibits`.
- Phase colors via `material.timeline.phase_for(code).text_class` (pr → `text-madras-green`, td → `text-madras-gold`, co → `text-madras-red`).
- `templates/base.html` and `templates/index.html` carry the owner's uncommitted edits: never modify or stage them; use explicit `git add` paths.
- Tests: `uv run python -m pytest -q` (49 pass at the start). Build CSS: `cd theme/static_src && npm run build`. Dev server: `uv run python manage.py runserver 8000` (`?djdt=hide`). Seeded chapter: `/exhibits/madras-in-south-india/`.

from exhibits.reader import reader_for
from exhibits.tests import make_chapter


class ReaderTest(TestCase):
    def setUp(self):
        self.home, self.pages = make_chapter("Chapter A", 1, ["One", "Two", "Three"])
        self.home2, self.pages2 = make_chapter("Chapter B", 2, ["Only"])

    def test_cover(self):
        r = reader_for(self.home)
        self.assertTrue(r.is_cover)
        self.assertIsNone(r.current)
        self.assertEqual([s.title for s in r.sections], ["One", "Two", "Three"])
        self.assertEqual((r.total, r.chapter_number, r.chapter_total), (3, 1, 2))
        self.assertEqual(r.next.title, "One")
        self.assertIsNone(r.previous)
        self.assertEqual(r.progress, 0.0)
        self.assertIsNone(r.previous_chapter)
        self.assertEqual(r.next_chapter, self.home2)
        self.assertEqual(r.reading_minutes, 1)  # tiny bodies floor to 1

    def test_middle_section(self):
        r = reader_for(self.pages[1])
        self.assertFalse(r.is_cover)
        self.assertEqual((r.current.number, r.current.title), (2, "Two"))
        self.assertEqual(r.previous.title, "One")
        self.assertEqual(r.next.title, "Three")
        self.assertAlmostEqual(r.progress, 2 / 3)
        self.assertEqual(r.current.url, self.pages[1].url)

    def test_ends(self):
        first, last = reader_for(self.pages[0]), reader_for(self.pages[2])
        self.assertIsNone(first.previous)
        self.assertIsNone(last.next)
        self.assertEqual(last.progress, 1.0)

    def test_second_chapter_and_unpublished_sections_are_skipped(self):
        self.pages2[0].unpublish()
        r = reader_for(self.home2)
        self.assertEqual((r.chapter_number, r.total), (2, 0))

        for page in (self.home, self.pages[0]):

Create `exhibits/reader.py`:

```python
"""Navigation and progress for the exhibit reader, derived from the page tree."""

from dataclasses import dataclass, field

from django.utils.html import strip_tags

WORDS_PER_MINUTE = 200


@dataclass
class Section:
    page: object
    number: int
    url: str
    title: str


@dataclass
class Reader:
    home: object
    chapter_number: int
    chapter_total: int
    sections: list[Section] = field(default_factory=list)
    total: int = 0
    reading_minutes: int = 1
    current: Section | None = None
    previous: Section | None = None
    next: Section | None = None
    progress: float = 0.0
    previous_chapter: object = None
    next_chapter: object = None

    @property
    def is_cover(self):
        return self.current is None


def reader_for(page):
    from exhibits.models import ExhibitHome, ExhibitPage

    if isinstance(page, ExhibitHome):
        home, current_page = page, None
    else:
        home, current_page = page.get_parent().specific, page

    section_pages = [
        p for p in home.get_children().live().specific() if isinstance(p, ExhibitPage)
    ]
    sections = [Section(p, i, p.url, p.title) for i, p in enumerate(section_pages, 1)]

    chapters = list(ExhibitHome.objects.live().order_by("exhibit_display_order", "path"))
    ids = [c.pk for c in chapters]
    idx = ids.index(home.pk) if home.pk in ids else 0
    words = sum(len(strip_tags(p.body or "").split()) for p in section_pages)

    reader = Reader(
        home=home,
        chapter_number=idx + 1,
        chapter_total=len(chapters),
        sections=sections,
        total=len(sections),
        reading_minutes=max(1, round(words / WORDS_PER_MINUTE)),
        previous_chapter=chapters[idx - 1] if idx > 0 else None,
        next_chapter=chapters[idx + 1] if idx + 1 < len(chapters) else None,
    )
    if current_page is None:
        reader.next = sections[0] if sections else None
        return reader

    position = next((i for i, s in enumerate(sections) if s.page.pk == current_page.pk), None)
    if position is None:  # unpublished or orphaned preview: behave like the cover
        reader.next = sections[0] if sections else None
        return reader
    reader.current = sections[position]
    reader.previous = sections[position - 1] if position > 0 else None
    reader.next = sections[position + 1] if position + 1 < len(sections) else None
    reader.progress = (position + 1) / len(sections)
    return reader
```
